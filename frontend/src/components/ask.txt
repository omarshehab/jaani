### 1. Backend — Fix draft response to include proper URLs

````javascript
const express = require('express');
const router = express.Router();
const { google } = require('googleapis');
const { getOAuth2Client } = require('../utils/oauth2client');

// Check Gmail connection status
router.get('/status', (req, res) => {
  try {
    const oauth2Client = getOAuth2Client();
    const tokens = oauth2Client.credentials;
    if (tokens && tokens.access_token) {
      res.json({ connected: true });
    } else {
      res.json({ connected: false });
    }
  } catch (err) {
    res.json({ connected: false });
  }
});

// Send email
router.post('/send', async (req, res) => {
  try {
    const { to, subject, body, cc, bcc } = req.body;
    const oauth2Client = getOAuth2Client();
    const gmail = google.gmail({ version: 'v1', auth: oauth2Client });

    // Build RFC 2822 message
    const messageParts = [
      `To: ${to}`,
      cc ? `Cc: ${cc}` : '',
      bcc ? `Bcc: ${bcc}` : '',
      `Subject: ${subject}`,
      'MIME-Version: 1.0',
      'Content-Type: text/html; charset=utf-8',
      '',
      body
    ].filter(Boolean);

    const rawMessage = messageParts.join('\r\n');
    const encodedMessage = Buffer.from(rawMessage)
      .toString('base64')
      .replace(/\+/g, '-')
      .replace(/\//g, '_')
      .replace(/=+$/, '');

    const result = await gmail.users.messages.send({
      userId: 'me',
      requestBody: { raw: encodedMessage }
    });

    res.json({
      success: true,
      messageId: result.data.id,
      message: 'Email sent successfully'
    });
  } catch (error) {
    console.error('Send email error:', error.message);
    res.status(500).json({ success: false, error: error.message });
  }
});

// Save draft
router.post('/draft', async (req, res) => {
  try {
    const { to, subject, body, cc, bcc } = req.body;
    const oauth2Client = getOAuth2Client();
    const gmail = google.gmail({ version: 'v1', auth: oauth2Client });

    // Build RFC 2822 message
    const messageParts = [
      to ? `To: ${to}` : '',
      cc ? `Cc: ${cc}` : '',
      bcc ? `Bcc: ${bcc}` : '',
      `Subject: ${subject || '(no subject)'}`,
      'MIME-Version: 1.0',
      'Content-Type: text/html; charset=utf-8',
      '',
      body || ''
    ].filter(Boolean);

    const rawMessage = messageParts.join('\r\n');
    const encodedMessage = Buffer.from(rawMessage)
      .toString('base64')
      .replace(/\+/g, '-')
      .replace(/\//g, '_')
      .replace(/=+$/, '');

    const result = await gmail.users.drafts.create({
      userId: 'me',
      requestBody: {
        message: { raw: encodedMessage }
      }
    });

    const draftId = result.data.id;
    const messageId = result.data.message?.id;

    // Compose URL for opening the draft directly in Gmail
    const composeUrl = `https://mail.google.com/mail/u/0/#drafts?compose=${messageId}`;
    const draftsUrl = `https://mail.google.com/mail/u/0/#drafts`;

    res.json({
      success: true,
      draftId,
      messageId,
      composeUrl,
      draftsUrl,
      message: 'Draft saved successfully'
    });
  } catch (error) {
    console.error('Save draft error:', error.message);
    res.status(500).json({ success: false, error: error.message });
  }
});
````

### 2. Home.js — Fix draft/send with pre-opened popup + OAuth queue

````javascript
import React, { useState, useRef, useCallback, useEffect } from "react";
import MailCard from "../components/MailCard";

const BACKEND = process.env.REACT_APP_BACKEND_URL || "http://localhost:5000";

export default function Home() {
  const [gmailConnected, setGmailConnected] = useState(false);
  const pendingActionRef = useRef(null);

  // Check Gmail connection on mount + after OAuth returns
  const checkGmailStatus = useCallback(async () => {
    try {
      const r = await fetch(`${BACKEND}/api/gmail/status`);
      const d = await r.json();
      setGmailConnected(d.connected);
      return d.connected;
    } catch {
      setGmailConnected(false);
      return false;
    }
  }, []);

  useEffect(() => {
    checkGmailStatus();
  }, [checkGmailStatus]);

  // Listen for OAuth callback completing in another tab
  useEffect(() => {
    const onFocus = async () => {
      const connected = await checkGmailStatus();
      if (connected && pendingActionRef.current) {
        const action = pendingActionRef.current;
        pendingActionRef.current = null;
        // Re-run the queued action
        if (action.type === "draft") {
          executeDraft(action.payload);
        } else if (action.type === "send") {
          executeSend(action.payload);
        }
      }
    };
    window.addEventListener("focus", onFocus);
    return () => window.removeEventListener("focus", onFocus);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  const startOAuth = () => {
    window.open(`${BACKEND}/api/auth/google`, "_blank", "width=500,height=600");
  };

  // ─── Draft ────────────────────────────────────────────
  const executeDraft = async (payload) => {
    try {
      const r = await fetch(`${BACKEND}/api/gmail/draft`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
      });
      const d = await r.json();
      if (d.success) {
        // Open drafts page in new tab
        const url = d.composeUrl || d.draftsUrl || "https://mail.google.com/mail/u/0/#drafts";
        window.open(url, "_blank");
        return { success: true };
      } else {
        return { success: false, error: d.error };
      }
    } catch (err) {
      return { success: false, error: err.message };
    }
  };

  const handleSaveDraft = async (payload) => {
    const connected = await checkGmailStatus();
    if (!connected) {
      pendingActionRef.current = { type: "draft", payload };
      startOAuth();
      return { success: false, error: "Connecting to Gmail... please authorize, then it will save automatically." };
    }
    return executeDraft(payload);
  };

  // ─── Send ─────────────────────────────────────────────
  const executeSend = async (payload) => {
    try {
      const r = await fetch(`${BACKEND}/api/gmail/send`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
      });
      const d = await r.json();
      if (d.success) {
        return { success: true };
      } else {
        return { success: false, error: d.error };
      }
    } catch (err) {
      return { success: false, error: err.message };
    }
  };

  const handleSendMail = async (payload) => {
    const connected = await checkGmailStatus();
    if (!connected) {
      pendingActionRef.current = { type: "send", payload };
      startOAuth();
      return { success: false, error: "Connecting to Gmail... please authorize, then it will send automatically." };
    }
    return executeSend(payload);
  };

  return (
    <div>
      <MailCard
        gmailConnected={gmailConnected}
        onSaveDraft={handleSaveDraft}
        onSendMail={handleSendMail}
        onConnectGmail={startOAuth}
      />
    </div>
  );
}
````

````jsx
import React, { useState } from "react";

/* ───────── Gmail-style Compose Card ───────── */
export default function MailCard({ gmailConnected, onSaveDraft, onSendMail, onConnectGmail }) {
  const [to, setTo] = useState("");
  const [cc, setCc] = useState("");
  const [bcc, setBcc] = useState("");
  const [subject, setSubject] = useState("");
  const [body, setBody] = useState("");
  const [showCc, setShowCc] = useState(false);
  const [showBcc, setShowBcc] = useState(false);
  const [status, setStatus] = useState(null); // { type: 'success'|'error'|'loading', msg }

  const clearStatus = () => setTimeout(() => setStatus(null), 4000);

  const payload = () => ({ to, cc, bcc, subject, body });

  const handleSend = async () => {
    if (!to.trim()) { setStatus({ type: "error", msg: "Please add a recipient." }); clearStatus(); return; }
    setStatus({ type: "loading", msg: "Sending…" });
    const res = await onSendMail(payload());
    if (res.success) {
      setStatus({ type: "success", msg: "Email sent ✓" });
      setTo(""); setCc(""); setBcc(""); setSubject(""); setBody("");
    } else {
      setStatus({ type: "error", msg: res.error || "Send failed" });
    }
    clearStatus();
  };

  const handleDraft = async () => {
    setStatus({ type: "loading", msg: "Saving draft…" });
    const res = await onSaveDraft(payload());
    if (res.success) {
      setStatus({ type: "success", msg: "Draft saved ✓ — opened in Gmail" });
    } else {
      setStatus({ type: "error", msg: res.error || "Draft failed" });
    }
    clearStatus();
  };

  const handleDiscard = () => {
    setTo(""); setCc(""); setBcc(""); setSubject(""); setBody("");
    setShowCc(false); setShowBcc(false);
    setStatus(null);
  };

  return (
    <div style={styles.wrapper}>
      {/* ─── Header ─── */}
      <div style={styles.header}>
        <span style={styles.headerTitle}>New Message</span>
        <div style={styles.headerRight}>
          {!gmailConnected && (
            <button onClick={onConnectGmail} style={styles.connectBtn}>
              Connect Gmail
            </button>
          )}
          {gmailConnected && <span style={styles.connectedBadge}>● Connected</span>}
        </div>
      </div>

      {/* ─── Fields ─── */}
      <div style={styles.fields}>
        {/* To */}
        <div style={styles.fieldRow}>
          <label style={styles.fieldLabel}>To</label>
          <input
            style={styles.fieldInput}
            value={to}
            onChange={e => setTo(e.target.value)}
            placeholder="recipients@example.com"
          />
          <div style={styles.ccBccToggle}>
            {!showCc && <span style={styles.toggleLink} onClick={() => setShowCc(true)}>Cc</span>}
            {!showBcc && <span style={styles.toggleLink} onClick={() => setShowBcc(true)}>Bcc</span>}
          </div>
        </div>

        {/* Cc */}
        {showCc && (
          <div style={styles.fieldRow}>
            <label style={styles.fieldLabel}>Cc</label>
            <input style={styles.fieldInput} value={cc} onChange={e => setCc(e.target.value)} />
          </div>
        )}

        {/* Bcc */}
        {showBcc && (
          <div style={styles.fieldRow}>
            <label style={styles.fieldLabel}>Bcc</label>
            <input style={styles.fieldInput} value={bcc} onChange={e => setBcc(e.target.value)} />
          </div>
        )}

        {/* Subject */}
        <div style={styles.fieldRow}>
          <label style={styles.fieldLabel}>Subject</label>
          <input
            style={styles.fieldInput}
            value={subject}
            onChange={e => setSubject(e.target.value)}
            placeholder="Subject"
          />
        </div>
      </div>

      {/* ─── Body ─── */}
      <textarea
        style={styles.body}
        value={body}
        onChange={e => setBody(e.target.value)}
        placeholder="Compose email"
        rows={12}
      />

      {/* ─── Status Banner ─── */}
      {status && (
        <div style={{
          ...styles.statusBar,
          background: status.type === "success" ? "#e6f4ea" : status.type === "error" ? "#fce8e6" : "#e8f0fe",
          color: status.type === "success" ? "#137333" : status.type === "error" ? "#c5221f" : "#1967d2",
        }}>

        <button style={styles.sendBtn} onClick={handleSend}>Send</button>
        <button style={styles.draftBtn} onClick={handleDraft} title="Save as Gmail draft">
          Save Draft
        </button>

        <button style={styles.discardBtn} onClick={handleDiscard} title="Discard">
          🗑
        </button>
      </div>
    </div>
  );
}
````
